\subsection{Step 4 --- Compensation / transfer function}
\subsubsection{4.0 Signal path}
NMCF: outer Miller $C_0$ from $v_1{=}$\texttt{net050} to $v_{\mathrm{out}}$;
inner Miller $C_1$ from $v_2{=}$\texttt{net049} to $v_{\mathrm{out}}$;
FF $g_{mf}$ from $v_1$ to $v_{\mathrm{out}}$. Differential drive into stage~1.

\subsubsection{4.1 Small-signal model}
Retain $g_{m1}v_{\mathrm{id}}$, $g_{m2}v_1$, $g_{m3}v_2$, $g_{mf}v_1$;
lumped $R_1,R_2,R_3$; caps $C_0,C_1,C_L$. Neglect device parasitics near GBW.

\subsubsection{4.2 Why each singularity}
\begin{itemize}
  \item $\omega_t{\approx}g_{m1}/C_0$: outer loop integrates via $C_0$ charging by stage~1.
  \item $\omega_{p1}{=}\omega_t/A_0$: dominant pole implied by DC gain and GBW.
  \item $\omega_{p2}{\approx}(g_{m2}/C_L)(C_0/C_1)$: nested ratio from solving the
        $v_2$/$v_{\mathrm{out}}$ KCL under high inner-loop gain (see 4.3).
  \item $\omega_{p3}{\approx}g_{m3}/C_1$: inner node against $C_1$.
  \item $\omega_z{\approx}g_{mf}/C_0$: LHP zero from FF current through $C_0$ path.
\end{itemize}

\subsubsection{4.3 Derive reduced $A(s)$ (per-node KCL)}
Nodes $v_1{=}$\texttt{net050}, $v_2{=}$\texttt{net049}, $v_{\mathrm{out}}$:
\begin{align*}
v_1&: \quad
  g_{m1} v_{\mathrm{id}} + \frac{v_1}{R_1}
  + s C_0 (v_1 - v_{\mathrm{out}}) = 0,\\
v_2&: \quad
  g_{m2} v_1 + \frac{v_2}{R_2}
  + s C_1 (v_2 - v_{\mathrm{out}}) = 0,\\
v_{\mathrm{out}}&: \quad
  g_{m3} v_2 + g_{mf} v_1 + \frac{v_{\mathrm{out}}}{R_3}
  + s C_L v_{\mathrm{out}}
  + s C_0 (v_{\mathrm{out}}-v_1)
  + s C_1 (v_{\mathrm{out}}-v_2) = 0.
\end{align*}
Approximations (separate lines):
\begin{enumerate}
  \item $A_{v2}A_{v3}\gg 1$ $\Rightarrow$ outer loop integrates with $\omega_t\approx g_{m1}/C_0$.
  \item Neglect device $C_{gs}/C_{gd}$ vs $C_0,C_1,C_L$ near GBW.
  \item Drop $s^2$ cross terms below GBW $\Rightarrow$ factor into 3p1z.
  \item Nested Miller: solving $v_2$ KCL with $v_{\mathrm{out}}\approx - (g_{m3}R_3)v_2$
        and substituting into the load equation yields the $C_0/C_1$ factor in $\omega_{p2}$.
\end{enumerate}
Reduced TF:
\[
A(s)=A_0\frac{1+s/\omega_z}{(1+s/\omega_{p1})(1+s/\omega_{p2})(1+s/\omega_{p3})}.
\]

\subsubsection{4.4 Named formulas}
\begin{align*}
\omega_t &= g_{m1}/C_0,\\
\omega_{p1} &= \omega_t/A_0,\\
\omega_{p2} &= (g_{m2}/C_L)\,(C_0/C_1),\\
\omega_{p3} &= g_{m3}/C_1,\\
\omega_z &= g_{mf}/C_0.
\end{align*}

\subsection{Step 5 --- GBW and poles/zeros (numeric)}

\end{align*}
